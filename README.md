#### **Day 3-5: Basic Player Movement (Mario Clone)**

**src/scenes/Level1Scene.js:**
```javascript
class Level1Scene extends Phaser.Scene {
    constructor() {
        super({ key: 'Level1Scene' });
    }

    preload() {
        // Load assets
        this.load.image('player', 'assets/sprites/modi-player.png');
        this.load.image('platform', 'assets/sprites/platforms.png');
        this.load.image('teaCup', 'assets/sprites/tea-cups.png');
    }

    create() {
        // Create platforms
        const platform = this.physics.add.staticImage(400, 580, 'platform');
        platform.setDisplaySize(800, 40);

        // Create player (Modi character)
        this.player = this.physics.add.sprite(100, 450, 'player');
        this.player.setBounce(0.1);
        this.player.setCollideWorldBounds(true);
        
        // Player physics
        this.player.body.setDrag(100, 0);
        this.player.body.setMaxVelocity(300, 600);

        // Collectibles (tea cups)
        this.teaCups = this.physics.add.group({
            key: 'teaCup',
            repeat: 15,
            setXY: { x: 100, y: 0, stepX: 40 }
        });

        this.teaCups.children.iterate(child => {
            child.setBounceY(0.3);
        });

        // Collisions
        this.physics.add.collider(this.player, platform);
        this.physics.add.collider(this.teaCups, platform);

        // Score
        this.score = 0;
        this.scoreText = this.add.text(16, 16, 'Tea Cups: 0', {
            fontSize: '24px',
            fill: '#fff'
        });

        // Controls
        this.cursors = this.input.keyboard.createCursorKeys();
    }

    update() {
        // Left/Right movement
        if (this.cursors.left.isDown) {
            this.player.setVelocityX(-200);
        } else if (this.cursors.right.isDown) {
            this.player.setVelocityX(200);
        } else {
            this.player.setVelocityX(0);
        }

        // Jump
        if (this.cursors.up.isDown && this.player.body.touching.down) {
            this.player.setVelocityY(-500);
        }

        // Collect tea cups
        this.physics.add.overlap(this.player, this.teaCups, (player, cup) => {
            cup.disableBody(true, true);
            this.score += 1;
            this.scoreText.setText('Tea Cups: ' + this.score);

class Enemy extends Phaser.Physics.Arcade.Sprite {
    constructor(scene, x, y, texture) {
        super(scene, x, y, texture);
        scene.physics.add.existing(this);
        
        this.setVelocityX(100);
        this.setCollideWorldBounds(true);
        
        // Bounce at edges
        this.scene.events.on('update', () => {
            if (this.body.velocity.x > 0 && this.x > this.scene.width - 30) {
                this.setVelocityX(-100);
            } else if (this.body.velocity.x < 0 && this.x < 30) {
                this.setVelocityX(100);
            }
        });
    }
}

// In LevelScene create():
this.enemies = this.physics.add.group();
const enemy = new Enemy(this, 600, 450, 'vendor');
this.enemies.add(enemy);

// Collision with player
this.physics.add.collider(this.player, this.enemies, (player, enemy) => {
    enemy.disableBody(true, true);
    player.setTint(0xff0000);
    this.score -= 5;

#### **Day 4-5: Boss Fight System**

**Boss: RSS Recruiter (Level 1 Boss)**

class Boss extends Phaser.Physics.Arcade.Sprite {
    constructor(scene, x, y) {
        super(scene, x, y, 'rss-recruiter');
        scene.physics.add.existing(this);
        
        this.health = 100;
        this.setImmuneToGravity();
        thissetDisplaySize(80, 80);
    }
    
    attack(player) {
        // Boss shoots "RSS pledge" projectiles
        const projectile = this.scene.physics.add.sprite(this.x, this.y, 'pledge');
        projectile.setVelocityX(-300);
        
        this.scene.physics.add.overlap(player, projectile, (p, proj) => {
            proj.disableBody(true, true);
            p.setTint(0xff0000);
            this.health -= 10;
        });
    }
}

// Boss spawn
this.boss = new Boss(this, 700, 400);
this.bossTimer = this.time.addEvent({
    delay: 2000,
    callback: () => this.boss.attack(this.player),
    loop: true
});

// Boss defeat
this.physics.add.overlap(this.player, this.boss, (player, boss) => {
    if (player.isInvincible) {
        boss.health -= 10;
        boss.setTint(0xff0000);
        
        if (boss.health <= 0) {
            boss.destroy();
            this.bossTimer.destroy();
            this.scene.start('Level2Scene'); // Next level
        }
    }
});
```
